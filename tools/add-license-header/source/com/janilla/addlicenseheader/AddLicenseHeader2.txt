	public static void main(String[] args) throws Exception {
		var h = Map.<String, Object>ofEntries(Map.entry("janilla", Map.ofEntries(Map.entry("", GPL2.split("\n")),
				Map.entry(Path.of("examples/acme-dashboard"), MIT_ACME_DASHBOARD.split("\n")),
				Map.entry(Path.of("examples/address-book"), MIT_ADDRESS_BOOK.split("\n")),
				Map.entry(Path.of("examples/com"), MIT.split("\n")),
				Map.entry(Path.of("examples/conduit"), MIT_CONDUIT.split("\n")),
				Map.entry(Path.of("examples/new-blank"), MIT_PAYLOAD.split("\n")),
				Map.entry(Path.of("examples/new-ecommerce"), MIT_PAYLOAD.split("\n")),
				Map.entry(Path.of("examples/new-website"), MIT_PAYLOAD.split("\n")),
				Map.entry(Path.of("examples/petclinic"), APACHE2_PETCLINIC.split("\n")),
				Map.entry(Path.of("examples/todomvc"), MIT_TODOMVC.split("\n")),
				Map.entry(Path.of("libraries/backend/source/com/janilla/backend/cms"), GPL2_PAYLOAD.split("\n")),
				Map.entry(Path.of("libraries/base/source/com/janilla/cms"), GPL2_PAYLOAD.split("\n")),
				Map.entry(Path.of("libraries/frontend/source/com/janilla/frontend/cms"), GPL2_PAYLOAD.split("\n")),
				Map.entry(Path.of("templates/blank"), MIT_PAYLOAD.split("\n")),
				Map.entry(Path.of("templates/ecommerce"), MIT_PAYLOAD.split("\n")),
				Map.entry(Path.of("templates/website"), MIT_PAYLOAD.split("\n")))));
		var s = Path.of("").resolve("../../..").toAbsolutePath().normalize();
//		IO.println(s);
		Files.walkFileTree(s, new SimpleFileVisitor<>() {

			@Override
			public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) throws IOException {
//				IO.println(dir);
				var n = dir.getFileName().toString();
				if (dir.getParent().equals(s)) {
					if (!h.containsKey(n))
						return FileVisitResult.SKIP_SUBTREE;
				}
				if (n.startsWith(".") || n.equals("src") || n.equals("bin") || n.equals("target"))
					return FileVisitResult.SKIP_SUBTREE;
				return FileVisitResult.CONTINUE;
			}

			@Override
			public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) throws IOException {
				var n = file.getFileName().toString();
				if (!n.endsWith(".css") && !n.endsWith(".java") && !n.endsWith(".js"))
					return FileVisitResult.CONTINUE;
//				IO.println(file);
				var l = Files.readAllLines(file);
				var i = l.iterator();
				var c = new Consumer<String>() {

					int state;

					@Override
					public void accept(String t) {
						state = switch (state) {
						case 0 -> t.equals("/*") ? 1 : -1;
						case 1 -> t.startsWith(" *") ? t.equals(" */") ? 2 : 1 : -1;
						case 2 -> 3;
						default -> throw new RuntimeException();
						};
					}
				};
				var a = Stream.iterate(i.hasNext() ? i.next() : null, t -> {
					if (t == null)
						return false;
					c.accept(t);
					return c.state == 1 || c.state == 2;
				}, _ -> i.hasNext() ? i.next() : null).toArray(String[]::new);
				var f = s.relativize(file);
				var o = h.get(f.getName(0).toString());
				String[] b;
				if (o instanceof Map<?, ?> m) {
					b = (String[]) m.get("");
					var x = 0;
					for (var e : m.entrySet())
						if (e.getKey() instanceof Path k && k.getNameCount() > x
								&& f.getName(0).relativize(f).startsWith(k)) {
							b = (String[]) e.getValue();
							x = k.getNameCount();
						}
				} else
					b = (String[]) o;
				if (!Arrays.equals(a, b)) {
					IO.println(file);
					Files.write(file, Stream.concat(Arrays.stream(b), l.stream().skip(a.length)).toList());
				}
				return FileVisitResult.CONTINUE;
			}
		});
	}
